-- ---------------------------------------------------------------------
-- CONSULTA 1: Recaudo de administración por torre
-- Problema de negocio: la administración quiere saber qué torres aportan
-- más dinero y cuáles tienen apartamentos que casi no pagan.
--
-- Qué hace:
--  * INNER JOIN torre -> apartamento (solo apartamentos que pertenecen a una torre).
--  * LEFT JOIN a pago, para que los apartamentos sin pagos no desaparezcan.
--    El filtro de fechas va en el ON del LEFT JOIN (y no en el WHERE) para no
--    convertirlo, sin querer, en un INNER JOIN.
--  * COUNT(DISTINCT ...) cuenta apartamentos totales y apartamentos que sí pagaron.
--  * SUM y AVG calculan el total recaudado y el pago promedio.
--  * HAVING deja solo las torres con al menos 4 pagos registrados.
-- ---------------------------------------------------------------------
SELECT  t.nombre_torre,
        COUNT(DISTINCT a.id_apartamento)              AS total_apartamentos,
        COUNT(DISTINCT p.id_apartamento)              AS apartamentos_con_pagos,
        COUNT(p.id_pago)                              AS num_pagos,
        COALESCE(SUM(p.monto_pagado), 0)              AS total_recaudado,
        ROUND(AVG(p.monto_pagado), 2)                 AS pago_promedio
FROM torre t
INNER JOIN apartamento a ON a.id_torre = t.id_torre
LEFT  JOIN pago p        ON p.id_apartamento = a.id_apartamento
                        AND p.fecha_pago BETWEEN DATE '2025-01-01' AND CURRENT_DATE
GROUP BY t.id_torre, t.nombre_torre
HAVING COUNT(p.id_pago) >= 4
ORDER BY total_recaudado DESC;


-- ---------------------------------------------------------------------
-- CONSULTA 2: Apartamentos morosos (cartera vencida)
-- Problema de negocio: generar la lista de cobro con los apartamentos que
-- tienen muchas cuotas vencidas sin ningún pago registrado.
--
-- Qué hace:
--  * CROSS JOIN apartamento x cuota_admin genera todas las combinaciones
--    "apartamento debe la cuota X".
--  * LEFT JOIN a pago y el filtro "p.id_pago IS NULL" dejan solo las
--    combinaciones SIN pago (patrón anti-join).
--  * WHERE c.fecha_vencimiento < CURRENT_DATE descarta cuotas que aún no vencen.
--  * INNER JOIN a residente (propietario o arrendatario) para tener un
--    responsable de contacto, y a torre para ubicar el apartamento.
--  * COUNT, SUM, MIN agrupan por apartamento: cuántas cuotas debe, cuánto
--    dinero es y desde qué periodo.
--  * HAVING solo muestra a quienes deben 12 cuotas o más; LIMIT trae los 15 peores.
-- Nota: un pago parcial cuenta como pagado; el saldo parcial se ve en la consulta 3.
-- ---------------------------------------------------------------------
SELECT  t.nombre_torre,
        a.numero_apto,
        r.nombres || ' ' || r.apellidos        AS responsable,
        r.rol,
        COUNT(c.id_cuota)                      AS cuotas_pendientes,
        SUM(c.valor)                           AS deuda_total,
        MIN(c.periodo)                         AS periodo_mas_antiguo,
        CURRENT_DATE - MIN(c.fecha_vencimiento) AS dias_mora_maxima
FROM apartamento a
INNER JOIN torre t     ON t.id_torre = a.id_torre
INNER JOIN residente r ON r.id_apartamento = a.id_apartamento
                      AND r.rol IN ('Propietario', 'Arrendatario')
CROSS JOIN cuota_admin c
LEFT  JOIN pago p      ON p.id_apartamento = a.id_apartamento
                      AND p.id_cuota = c.id_cuota
WHERE c.fecha_vencimiento < CURRENT_DATE
  AND p.id_pago IS NULL
GROUP BY t.nombre_torre, a.id_apartamento, a.numero_apto, r.id_residente, r.nombres, r.apellidos, r.rol
HAVING COUNT(c.id_cuota) >= 12
ORDER BY deuda_total DESC, dias_mora_maxima DESC
LIMIT 15;


-- ---------------------------------------------------------------------
-- CONSULTA 3: Comportamiento de pago por periodo
-- Problema de negocio: detectar en qué meses se pagó tarde o de forma
-- incompleta, para decidir cuándo reforzar los recordatorios de cobro.
--
-- Qué hace:
--  * INNER JOIN cuota_admin -> pago -> apartamento (solo cuotas que ya tienen pagos).
--  * SUM(CASE WHEN ...) cuenta condicionalmente: pagos tardíos
--    (fecha_pago > fecha_vencimiento) y pagos parciales (monto < valor de la cuota).
--  * SUM(c.valor - p.monto_pagado) calcula el saldo que quedó sin pagar
--    en los pagos parciales.
--  * AVG(p.fecha_pago - c.fecha_vencimiento) da los días promedio respecto
--    al vencimiento (negativo = pagó antes, positivo = pagó después).
--  * HAVING deja solo los periodos con al menos un pago tardío o parcial.
-- ---------------------------------------------------------------------
SELECT  c.periodo,
        c.valor                                              AS valor_cuota,
        COUNT(p.id_pago)                                     AS pagos_recibidos,
        COUNT(DISTINCT a.id_torre)                           AS torres_que_pagaron,
        SUM(CASE WHEN p.fecha_pago > c.fecha_vencimiento THEN 1 ELSE 0 END) AS pagos_tardios,
        SUM(CASE WHEN p.monto_pagado < c.valor THEN 1 ELSE 0 END)           AS pagos_parciales,
        SUM(p.monto_pagado)                                  AS total_recaudado,
        SUM(c.valor - p.monto_pagado)                        AS saldo_sin_pagar,
        ROUND(AVG(p.fecha_pago - c.fecha_vencimiento), 1)    AS dias_promedio_vs_vencimiento
FROM cuota_admin c
INNER JOIN pago p        ON p.id_cuota = c.id_cuota
INNER JOIN apartamento a ON a.id_apartamento = p.id_apartamento
GROUP BY c.id_cuota, c.periodo, c.valor
HAVING SUM(CASE WHEN p.fecha_pago > c.fecha_vencimiento THEN 1 ELSE 0 END) >= 1
    OR SUM(CASE WHEN p.monto_pagado < c.valor THEN 1 ELSE 0 END) >= 1
ORDER BY c.periodo;


-- ---------------------------------------------------------------------
-- CONSULTA 4: Uso e ingresos de las zonas comunes
-- Problema de negocio: saber qué zonas se usan más, cuántas horas se
-- reservan y cuánto dinero generan, para decidir mantenimiento o cambios de tarifa.
--
-- Qué hace:
--  * LEFT JOIN zona_comun -> reserva: las zonas que nunca se reservan
--    también aparecen (con 0). El filtro de estado va en el ON para no perderlas.
--  * Solo cuenta reservas "Finalizada" o "Confirmada" (ignora canceladas y pendientes).
--  * EXTRACT(EPOCH FROM hora_fin - hora_inicio)/3600 convierte la duración a horas.
--  * COUNT, SUM y AVG dan reservas, horas totales y duración promedio.
--  * Ingresos = horas reservadas x tarifa_hora de la zona (solo las BBQ cobran).
--  * CASE clasifica cada zona según su nivel de uso.
-- ---------------------------------------------------------------------
SELECT  z.nombre_zona,
        z.tarifa_hora,
        COUNT(r.id_reserva)                                                         AS reservas_efectivas,
        COALESCE(ROUND(SUM(EXTRACT(EPOCH FROM (r.hora_fin - r.hora_inicio)) / 3600)::numeric, 1), 0)
                                                                                    AS horas_reservadas,
        ROUND(AVG(EXTRACT(EPOCH FROM (r.hora_fin - r.hora_inicio)) / 3600)::numeric, 2)
                                                                                    AS duracion_promedio_h,
        COALESCE(ROUND(SUM(EXTRACT(EPOCH FROM (r.hora_fin - r.hora_inicio)) / 3600 * z.tarifa_hora)::numeric, 2), 0)
                                                                                    AS ingresos_generados,
        CASE WHEN COUNT(r.id_reserva) >= 4 THEN 'Alto uso'
             WHEN COUNT(r.id_reserva) >= 2 THEN 'Uso medio'
             ELSE 'Subutilizada' END                                                AS nivel_uso
FROM zona_comun z
LEFT JOIN reserva r ON r.id_zona = z.id_zona
                   AND r.estado IN ('Finalizada', 'Confirmada')
GROUP BY z.id_zona, z.nombre_zona, z.tarifa_hora
ORDER BY reservas_efectivas DESC, ingresos_generados DESC;


-- ---------------------------------------------------------------------
-- CONSULTA 5: Residentes que más reservan zonas comunes
-- Problema de negocio: identificar a los residentes más activos en el uso
-- de zonas comunes y cuánto han gastado en zonas con tarifa.
--
-- Qué hace:
--  * INNER JOIN de cinco tablas: residente -> apartamento -> torre
--    y residente -> reserva -> zona_comun.
--  * WHERE filtra residentes activos (estado = TRUE) y reservas no canceladas.
--  * COUNT(*) FILTER (WHERE ...) cuenta solo las reservas en zonas con tarifa.
--  * SUM(horas x tarifa) calcula el gasto total y AVG la duración promedio.
--  * HAVING deja a quienes tienen 2 o más reservas.
-- ---------------------------------------------------------------------
SELECT  r.id_residente,
        r.nombres || ' ' || r.apellidos                    AS residente,
        t.nombre_torre || ' - Apto ' || a.numero_apto      AS vivienda,
        COUNT(rs.id_reserva)                               AS total_reservas,
        COUNT(*) FILTER (WHERE z.tarifa_hora > 0)          AS reservas_con_tarifa,
        COUNT(DISTINCT z.id_zona)                          AS zonas_distintas,
        ROUND(AVG(EXTRACT(EPOCH FROM (rs.hora_fin - rs.hora_inicio)) / 3600)::numeric, 2) AS duracion_promedio_h,
        COALESCE(ROUND(SUM(EXTRACT(EPOCH FROM (rs.hora_fin - rs.hora_inicio)) / 3600 * z.tarifa_hora)::numeric, 2), 0)
                                                           AS gasto_total
FROM residente r
INNER JOIN apartamento a ON a.id_apartamento = r.id_apartamento
INNER JOIN torre t       ON t.id_torre = a.id_torre
INNER JOIN reserva rs    ON rs.id_residente = r.id_residente
INNER JOIN zona_comun z  ON z.id_zona = rs.id_zona
WHERE r.estado = TRUE
  AND rs.estado <> 'Cancelada'
GROUP BY r.id_residente, r.nombres, r.apellidos, t.nombre_torre, a.numero_apto
HAVING COUNT(rs.id_reserva) >= 2
ORDER BY total_reservas DESC, gasto_total DESC;


-- ---------------------------------------------------------------------
-- CONSULTA 6: Ocupación de parqueaderos y vehículos por torre
-- Problema de negocio: ver qué torres tienen déficit de parqueaderos y
-- cuántos apartamentos no tienen ninguno asignado.
--
-- Qué hace:
--  * INNER JOIN torre -> apartamento, y LEFT JOIN a parqueadero y vehiculo,
--    porque no todos los apartamentos tienen parqueadero ni todos los
--    parqueaderos tienen vehículo.
--  * COUNT(DISTINCT ...) evita contar doble por la multiplicación de filas de los joins.
--  * FILTER separa parqueaderos de carro y de moto, y apartamentos sin parqueadero.
--  * El ratio muestra cuántos parqueaderos hay por apartamento.
--  * HAVING descarta torres sin ningún parqueadero.
-- ---------------------------------------------------------------------
SELECT  t.nombre_torre,
        COUNT(DISTINCT a.id_apartamento)                                                AS apartamentos,
        COUNT(DISTINCT pq.id_parqueadero)                                               AS parqueaderos,
        COUNT(DISTINCT pq.id_parqueadero) FILTER (WHERE pq.tipo_parqueadero = 'Carro')  AS parq_carro,
        COUNT(DISTINCT pq.id_parqueadero) FILTER (WHERE pq.tipo_parqueadero = 'Moto')   AS parq_moto,
        COUNT(v.id_vehiculo)                                                            AS vehiculos_registrados,
        COUNT(DISTINCT a.id_apartamento) FILTER (WHERE pq.id_parqueadero IS NULL)       AS aptos_sin_parqueadero,
        ROUND(COUNT(DISTINCT pq.id_parqueadero)::numeric / COUNT(DISTINCT a.id_apartamento), 2) AS parqueaderos_por_apto
FROM torre t
INNER JOIN apartamento a  ON a.id_torre = t.id_torre
LEFT  JOIN parqueadero pq ON pq.id_apartamento = a.id_apartamento
LEFT  JOIN vehiculo v     ON v.id_parqueadero = pq.id_parqueadero
GROUP BY t.id_torre, t.nombre_torre
HAVING COUNT(DISTINCT pq.id_parqueadero) > 0
ORDER BY parqueaderos_por_apto ASC;


-- ---------------------------------------------------------------------
-- CONSULTA 7: Visitantes frecuentes
-- Problema de negocio: seguridad quiere identificar visitantes recurrentes
-- (posibles proveedores o terceros habituales) y quién sigue dentro del conjunto.
--
-- Qué hace:
--  * INNER JOIN visitante -> ingreso_visita -> apartamento -> torre.
--  * WHERE limita el análisis a los ingresos desde el 1 de septiembre de 2026.
--  * COUNT cuenta visitas; COUNT(DISTINCT) cuenta apartamentos distintos visitados.
--  * AVG de (hora_salida - hora_entrada) da la permanencia promedio en minutos
--    (las visitas sin salida quedan fuera del promedio).
--  * FILTER cuenta las visitas con hora_salida NULL, o sea, quien sigue adentro.
--  * STRING_AGG junta las torres visitadas en un solo texto.
--  * HAVING deja solo visitantes con 2 o más ingresos.
-- ---------------------------------------------------------------------
SELECT  v.nombre_visitante,
        v.tipo_documento || ' ' || v.numero_documento                                    AS documento,
        COUNT(iv.id_ingreso)                                                             AS visitas,
        COUNT(DISTINCT iv.id_apartamento)                                                AS apartamentos_visitados,
        STRING_AGG(DISTINCT t.nombre_torre, ', ' ORDER BY t.nombre_torre)                AS torres_visitadas,
        ROUND(AVG(EXTRACT(EPOCH FROM (iv.hora_salida - iv.hora_entrada)) / 60)::numeric, 0) AS minutos_promedio,
        COUNT(*) FILTER (WHERE iv.hora_salida IS NULL)                                   AS aun_dentro,
        MAX(iv.hora_entrada)                                                             AS ultimo_ingreso
FROM visitante v
INNER JOIN ingreso_visita iv ON iv.id_visitante = v.id_visitante
INNER JOIN apartamento a     ON a.id_apartamento = iv.id_apartamento
INNER JOIN torre t           ON t.id_torre = a.id_torre
WHERE iv.hora_entrada >= TIMESTAMP '2026-09-01 00:00:00'
GROUP BY v.id_visitante, v.nombre_visitante, v.tipo_documento, v.numero_documento
HAVING COUNT(iv.id_ingreso) >= 2
ORDER BY visitas DESC, ultimo_ingreso DESC;


-- ---------------------------------------------------------------------
-- CONSULTA 8: Carga de trabajo de porteros y vigilantes
-- Problema de negocio: repartir mejor los turnos, mostrando quién registra
-- más ingresos que el promedio de sus compañeros.
--
-- Qué hace:
--  * LEFT JOIN trabajador -> ingreso_visita (incluye a quien no registró nada)
--    y LEFT JOIN a telefono para tener su contacto.
--  * WHERE se queda solo con los cargos Portero y Vigilante.
--  * COUNT(DISTINCT ...) es necesario porque unir con telefono repite las filas
--    de ingresos por cada teléfono del trabajador.
--  * AVG calcula la duración promedio de las visitas que registró.
--  * HAVING compara contra una subconsulta: el promedio de ingresos por
--    trabajador de toda la tabla ingreso_visita. Solo quedan los que están por encima.
-- ---------------------------------------------------------------------
SELECT  tr.nombres || ' ' || tr.apellidos                  AS trabajador,
        tr.cargo,
        COUNT(DISTINCT iv.id_ingreso)                      AS ingresos_registrados,
        COUNT(DISTINCT iv.id_visitante)                    AS visitantes_distintos,
        COUNT(DISTINCT iv.id_apartamento)                  AS apartamentos_atendidos,
        ROUND(AVG(EXTRACT(EPOCH FROM (iv.hora_salida - iv.hora_entrada)) / 60)::numeric, 0) AS minutos_promedio_visita,
        COUNT(DISTINCT tel.id_telefono)                    AS telefonos_contacto
FROM trabajador tr
LEFT JOIN ingreso_visita iv ON iv.id_trabajador = tr.id_trabajador
LEFT JOIN telefono tel      ON tel.id_trabajador = tr.id_trabajador
WHERE tr.cargo IN ('Portero', 'Vigilante')
GROUP BY tr.id_trabajador, tr.nombres, tr.apellidos, tr.cargo
HAVING COUNT(DISTINCT iv.id_ingreso) >
       (SELECT COUNT(*)::numeric / COUNT(DISTINCT id_trabajador) FROM ingreso_visita)
ORDER BY ingresos_registrados DESC;


-- ---------------------------------------------------------------------
-- CONSULTA 9: Auditoría de pagos por origen del registro
-- Problema de negocio: medir qué canal (app web, app móvil, trigger, manual)
-- genera más intentos de pago fallidos y cuánto dinero no se aplicó.
--
-- Qué hace:
--  * Parte de auditoria_pago y hace INNER JOIN a apartamento, torre y cuota_admin
--    (todos los registros tienen apartamento y cuota válidos).
--  * LEFT JOIN a pago por id_pago: id_pago no es FK, y en los intentos fallidos
--    es NULL, así que un INNER JOIN los eliminaría.
--  * FILTER cuenta exitosos y fallidos; la tasa de fallo es fallidos / total x 100.
--  * SUM(monto_intentado - monto_aplicado) es el dinero que se intentó cobrar y no se aplicó.
--  * La columna "conciliados" verifica que el monto auditado coincida con el pago real.
--  * STRING_AGG lista los códigos de error distintos de cada origen.
--  * HAVING deja solo los orígenes con al menos 3 registros.
-- ---------------------------------------------------------------------
SELECT  ap.origen,
        COUNT(*)                                                          AS total_registros,
        COUNT(*) FILTER (WHERE ap.resultado = 'EXITOSO')                  AS exitosos,
        COUNT(*) FILTER (WHERE ap.resultado = 'FALLIDO')                  AS fallidos,
        ROUND(100.0 * COUNT(*) FILTER (WHERE ap.resultado = 'FALLIDO') / COUNT(*), 1) AS tasa_fallo_pct,
        SUM(ap.monto_intentado - ap.monto_aplicado)                       AS monto_no_aplicado,
        COUNT(*) FILTER (WHERE p.id_pago IS NOT NULL
                           AND p.monto_pagado = ap.monto_aplicado)        AS conciliados_con_pago,
        COUNT(DISTINCT t.id_torre)                                        AS torres_involucradas,
        STRING_AGG(DISTINCT ap.codigo_error, ', ')                        AS codigos_error
FROM auditoria_pago ap
INNER JOIN apartamento a ON a.id_apartamento = ap.id_apartamento
INNER JOIN torre t       ON t.id_torre = a.id_torre
INNER JOIN cuota_admin c ON c.id_cuota = ap.id_cuota
LEFT  JOIN pago p        ON p.id_pago = ap.id_pago
WHERE ap.fecha_hora >= TIMESTAMP '2025-01-01 00:00:00'
GROUP BY ap.origen
HAVING COUNT(*) >= 3
ORDER BY tasa_fallo_pct DESC, total_registros DESC;


-- ---------------------------------------------------------------------
-- CONSULTA 10: Perfil integral de apartamentos con recaudo bajo el promedio
-- Problema de negocio: ver, en una sola tabla, a quién hay que priorizar
-- para gestión de cobro, combinando residentes, vehículos y pagos.
--
-- Qué hace:
--  * Usa tres CTEs (WITH) que agregan primero cada tabla por apartamento. Así
--    se evita el error clásico de multiplicar filas al unir residentes,
--    vehículos y pagos al mismo tiempo.
--      - pagos_apto: número de pagos y total pagado (GROUP BY).
--      - res_apto: residentes activos.
--      - veh_apto: vehículos, pasando por parqueadero (con HAVING para
--        incluir solo apartamentos que sí tienen vehículo).
--  * La consulta principal une apartamento -> torre con LEFT JOIN a los tres
--    CTEs, y usa COALESCE para mostrar 0 donde no hay datos.
--  * El WHERE usa una subconsulta: solo apartamentos cuyo total pagado es menor
--    al promedio por apartamento de todo el conjunto.
--  * pagado_por_m2 normaliza el recaudo según el tamaño del apartamento.
-- ---------------------------------------------------------------------
WITH pagos_apto AS (
    SELECT id_apartamento,
           COUNT(*)           AS num_pagos,
           SUM(monto_pagado)  AS total_pagado
    FROM pago
    GROUP BY id_apartamento
),
res_apto AS (
    SELECT id_apartamento,
           COUNT(*) FILTER (WHERE estado) AS residentes_activos
    FROM residente
    GROUP BY id_apartamento
),
veh_apto AS (
    SELECT pq.id_apartamento,
           COUNT(v.id_vehiculo) AS vehiculos
    FROM parqueadero pq
    INNER JOIN vehiculo v ON v.id_parqueadero = pq.id_parqueadero
    GROUP BY pq.id_apartamento
    HAVING COUNT(v.id_vehiculo) >= 1
)
SELECT  t.nombre_torre,
        a.numero_apto,
        a.area_m2,
        COALESCE(ra.residentes_activos, 0)                       AS residentes_activos,
        COALESCE(va.vehiculos, 0)                                AS vehiculos,
        COALESCE(pa.num_pagos, 0)                                AS num_pagos,
        COALESCE(pa.total_pagado, 0)                             AS total_pagado,
        ROUND(COALESCE(pa.total_pagado, 0) / a.area_m2, 2)       AS pagado_por_m2
FROM apartamento a
INNER JOIN torre t        ON t.id_torre = a.id_torre
LEFT  JOIN res_apto ra    ON ra.id_apartamento = a.id_apartamento
LEFT  JOIN veh_apto va    ON va.id_apartamento = a.id_apartamento
LEFT  JOIN pagos_apto pa  ON pa.id_apartamento = a.id_apartamento
WHERE COALESCE(pa.total_pagado, 0) <
      (SELECT SUM(monto_pagado) / (SELECT COUNT(*) FROM apartamento) FROM pago)
ORDER BY total_pagado ASC, residentes_activos DESC;
